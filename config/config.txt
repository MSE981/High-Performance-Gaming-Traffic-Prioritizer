# High-performance gaming traffic prioritizer - router configuration
# Place this file next to the GamingTrafficPrioritizer binary, or pass a path at startup.
# All keys are optional. The program starts with built-in defaults if this file is absent.
# The GUI writes changes back to this file on every clean exit.

# ---- Network interfaces ----
# IFACE_WAN  : Ethernet interface facing your router (internet side).
# IFACE_LAN  : Ethernet interface facing your local devices.
# ROUTER_IP  : The Pi's own IP on the LAN side (default gateway for DHCP clients).
#              Must be on a different subnet from the upstream router.
# APPLY_ROUTER_IP_TO_LAN : If true, the program sets the kernel IPv4 on IFACE_LAN to ROUTER_IP at startup
#              (userspace DHCP does not configure the interface). Requires root. Disable if you use netplan/systemd-networkd.
# LAN_PREFIX_LEN : CIDR prefix for that LAN (typical home LAN: 24).
APPLY_ROUTER_IP_TO_LAN=true
LAN_PREFIX_LEN=24
IFACE_WAN=eth0
IFACE_LAN=eth1
ROUTER_IP=192.168.12.1

# ---- QoS and traffic classification ----
# ENABLE_ACCELERATION    : true  = two-tier QoS, high priority fast lane
#                          false = transparent bridge, no classification
# LARGE_PACKET_THRESHOLD : Packets below this byte count (and DNS) go to the
#                          High (unthrottled) lane; larger packets go to Normal.
ENABLE_ACCELERATION=true
LARGE_PACKET_THRESHOLD=1000

# ---- Built-in services ----
# Each service can be toggled independently without restarting.
# The GUI Service page writes these values on exit.
enable_nat=true
enable_dhcp=true

# ---- DHCP address pool (LAN clients) ----
# Pool bounds imply the LAN subnet (prefix). ROUTER_IP must lie in that subnet; when enable_dhcp=true,
# the program aligns IFACE_LAN into that subnet (see APPLY_ROUTER_IP_TO_LAN) and uses the kernel LAN IP
# as the DHCP default gateway.
DHCP_POOL_START=192.168.12.50
DHCP_POOL_END=192.168.12.255
